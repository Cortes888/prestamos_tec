Trigger 1 – Al crear préstamo: cambiar objeto a 'prestado'
CREATE TRIGGER trg_prestamo_insert AFTER INSERT ON prestamo FOR EACH ROW BEGIN IF NEW.estado = 'activo' THEN UPDATE objeto SET estado = 'prestado' WHERE id_objeto = NEW.id_objeto; END IF; END
Trigger 2 – Al devolver: libera objeto + genera notificación si vence
CREATE TRIGGER trg_prestamo_devolucion AFTER UPDATE ON prestamo FOR EACH ROW BEGIN -- Si cambió a devuelto → liberar objeto IF NEW.estado = 'devuelto' AND OLD.estado != 'devuelto' THEN UPDATE objeto SET estado = 'disponible' WHERE id_objeto = NEW.id_objeto; END IF; -- Si cambió a vencido → insertar notificación automática IF NEW.estado = 'vencido' AND OLD.estado != 'vencido' THEN INSERT INTO notificacion (id_prestamo, correo_destino, asunto, cuerpo) SELECT NEW.id_prestamo, u.correo, CONCAT('⚠️ Devolución pendiente – ', o.nombre), '...' FROM usuario u JOIN objeto o ON o.id_objeto = NEW.id_objeto WHERE u.id_usuario = NEW.id_usuario; END IF; END
Trigger 3 – Historial de cambios de estado del objeto
CREATE TRIGGER trg_objeto_estado AFTER UPDATE ON objeto FOR EACH ROW BEGIN IF OLD.estado != NEW.estado THEN INSERT INTO historial_estado_objeto (id_objeto, estado_ant, estado_nuevo, motivo) VALUES (NEW.id_objeto, OLD.estado, NEW.estado, 'Cambio manual'); END IF; END

JOIN 1 – Préstamos con usuario, facultad, objeto y categoría
SELECT p.id_prestamo, CONCAT(u.nombre,' ',u.apellido) AS usuario, f.nombre AS facultad, o.nombre AS objeto, c.nombre AS categoria, p.estado FROM prestamo p JOIN usuario u ON u.id_usuario=p.id_usuario JOIN facultad f ON f.id_facultad=u.id_facultad JOIN objeto o ON o.id_objeto=p.id_objeto JOIN categoria c ON c.id_categoria=o.id_categoria;
JOIN 2 – Vencidos con días de retraso
SELECT u.correo, o.nombre AS objeto, fn_dias_retraso(p.id_prestamo) AS dias_retraso FROM prestamo p JOIN usuario u ON u.id_usuario=p.id_usuario JOIN objeto o ON o.id_objeto=p.id_objeto WHERE p.estado='vencido' ORDER BY dias_retraso DESC;
JOIN 3 – Notificaciones con detalle de préstamo
SELECT n.asunto, n.correo_destino, n.enviado, o.nombre AS objeto, CONCAT(u.nombre,' ',u.apellido) AS usuario FROM notificacion n JOIN prestamo p ON p.id_prestamo=n.id_prestamo JOIN objeto o ON o.id_objeto=p.id_objeto JOIN usuario u ON u.id_usuario=p.id_usuario;
JOIN 4 – Disponibilidad por categoría (GROUP BY)
SELECT c.nombre, COUNT(o.id_objeto) AS total, SUM(o.estado='disponible') AS disponibles FROM categoria c LEFT JOIN objeto o ON o.id_categoria=c.id_categoria GROUP BY c.id_categoria, c.nombre;
